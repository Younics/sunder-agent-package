using Microsoft.Extensions.DependencyInjection;
using Sunder.Package.Agent.Protocol;
using Sunder.Package.Agent.Provider.LMStudio;
using Sunder.Package.Agent.Provider.LMStudio.Protocol;
using Sunder.Package.Agent.Provider.LMStudio.Protocol.Generated;
using Sunder.Package.Agent.Provider.Shared;
using Sunder.Sdk.Abstractions;
using Sunder.Sdk.Rpc;
using Sunder.Sdk.Worker;

ServiceProvider? ownedServices = null;
try
{
    await SunderWorker.RunAsync(context =>
    {
        var services = new ServiceCollection();
        services.AddSingleton<IPackageContext>(context);
        services.AddSingleton(new ProviderCredentialAccessor(
            context.Secrets,
            LMStudioProviderConfiguration.ApiKeyKey));
        services.AddSingleton<ProviderCredentialRpcProvider>();
        services.AddSingleton(serviceProvider => new LMStudioConnection(
            context,
            serviceProvider.GetRequiredService<ProviderCredentialAccessor>()));
        services.AddSingleton(serviceProvider => new LMStudioModelCatalogService(
            serviceProvider.GetRequiredService<LMStudioConnection>()));
        services.AddSingleton(serviceProvider => new LMStudioAgentProvider(
            context,
            serviceProvider.GetRequiredService<LMStudioConnection>(),
            serviceProvider.GetRequiredService<LMStudioModelCatalogService>()));
        services.AddSingleton(serviceProvider => new LMStudioEmbeddingProvider(
            context,
            serviceProvider.GetRequiredService<LMStudioConnection>(),
            serviceProvider.GetRequiredService<LMStudioModelCatalogService>()));
        var serviceProvider = services.BuildServiceProvider(new ServiceProviderOptions
        {
            ValidateOnBuild = true,
            ValidateScopes = true,
        });
        ownedServices = serviceProvider;

        return new SunderWorkerOptions(
        [
            Register(
                "lmstudio.chat",
                AgentRpcContractDescriptors.Get(AgentRpcContractIds.ChatProvider),
                AgentChatProviderRpc.CreateHandler(serviceProvider.GetRequiredService<LMStudioAgentProvider>())),
            Register(
                "lmstudio.embedding",
                AgentRpcContractDescriptors.Get(AgentRpcContractIds.EmbeddingProvider),
                AgentEmbeddingProviderRpc.CreateHandler(serviceProvider.GetRequiredService<LMStudioEmbeddingProvider>())),
            Register(
                "lmstudio.credentials",
                LMStudioCredentialProtocol.Descriptor,
                new SunderPackageAgentProviderLmstudioCredentialsRpcHandler(
                    serviceProvider.GetRequiredService<ProviderCredentialRpcProvider>())),
        ])
        {
            SettingsSchema = LMStudioProviderConfiguration.Schema,
            OnShutdown = async (_, _) =>
            {
                await serviceProvider.DisposeAsync();
                if (ReferenceEquals(ownedServices, serviceProvider)) ownedServices = null;
            },
        };
    });
}
finally
{
    if (ownedServices is not null) await ownedServices.DisposeAsync();
}

static SunderWorkerProviderRegistration Register(
    string providerId,
    SunderRpcContractDescriptor descriptor,
    ISunderRpcServiceHandler handler)
    => new(
        providerId,
        descriptor.ContractId,
        descriptor.Version,
        descriptor.Sha256,
        descriptor.AbiSha256 ?? throw new InvalidOperationException(
            $"RPC contract '{descriptor.ContractId}' does not have an ABI identity."),
        handler);
