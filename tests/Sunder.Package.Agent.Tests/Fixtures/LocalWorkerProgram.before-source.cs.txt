using Microsoft.Extensions.DependencyInjection;
using Sunder.Package.Agent.Execution.Local;
using Sunder.Package.Agent.Protocol;
using Sunder.Sdk.Abstractions;
using Sunder.Sdk.Rpc;
using Sunder.Sdk.Worker;

SunderWorkerComposition? ownedComposition = null;
try
{
    await SunderWorker.RunAsync(context =>
    {
        var services = new ServiceCollection();
        services.AddSingleton<IPackageContext>(context);
        services.AddSingleton<LocalShellCatalogService>();
        services.AddSingleton<LocalExecutionWorkspaceConfigService>();
        services.AddSingleton(provider => new LocalExecutionTarget(
            context,
            provider.GetRequiredService<LocalExecutionWorkspaceConfigService>(),
            provider.GetRequiredService<LocalShellCatalogService>()));
        services.AddSingleton<LocalExecutionWorkspaceEditorContributor>();
        services.AddSingleton<LocalExecutionPresentationRpcHandler>();
        var serviceProvider = services.BuildServiceProvider(new ServiceProviderOptions
        {
            ValidateOnBuild = true,
            ValidateScopes = true,
        });
        var composition = new SunderWorkerComposition(serviceProvider);
        ownedComposition = composition;
        var target = serviceProvider.GetRequiredService<LocalExecutionTarget>();

        return new SunderWorkerOptions(
        [
            RegisterAgent(
                "local.execution.target",
                AgentRpcContractIds.ExecutionTarget,
                AgentExecutionTargetRpc.CreateHandler(target)),
            RegisterAgent(
                "local.workspace.path.migrator",
                AgentRpcContractIds.WorkspacePathMigrator,
                AgentWorkspacePathMigratorRpc.CreateHandler(
                    serviceProvider.GetRequiredService<LocalExecutionWorkspaceConfigService>())),
            RegisterAgent(
                "local.workspace.editor",
                AgentRpcContractIds.WorkspaceEditor,
                AgentWorkspaceEditorRpc.CreateHandler(
                    serviceProvider.GetRequiredService<LocalExecutionWorkspaceEditorContributor>())),
            Register(
                LocalExecutionPresentationRpc.ProviderId,
                LocalExecutionPresentationRpc.Descriptor,
                LocalExecutionPresentationRpc.CreateHandler(
                    serviceProvider.GetRequiredService<LocalExecutionPresentationRpcHandler>())),
        ])
        {
            SettingsSchema = LocalExecutionConfiguration.Schema,
            Composition = composition,
        };
    });
}
finally
{
    if (ownedComposition is not null)
    {
        await ownedComposition.DisposeAsync();
    }
}

static SunderWorkerProviderRegistration RegisterAgent(
    string providerId,
    string contractId,
    ISunderRpcServiceHandler handler)
    => Register(providerId, AgentRpcContractDescriptors.Get(contractId), handler);

static SunderWorkerProviderRegistration Register(
    string providerId,
    SunderRpcContractDescriptor descriptor,
    ISunderRpcServiceHandler handler)
    => new(
        providerId,
        descriptor.ContractId,
        descriptor.Version,
        descriptor.Sha256,
        descriptor.AbiSha256 ?? throw new InvalidOperationException(
            $"RPC contract '{descriptor.ContractId}' does not have an ABI identity."),
        handler);
