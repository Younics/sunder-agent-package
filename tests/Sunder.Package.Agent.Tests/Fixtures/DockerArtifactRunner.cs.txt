using Sunder.Package.Agent.Execution.Docker;
using Sunder.Sdk.Abstractions;
using System.Reflection;
using System.Text.Json;

public sealed class ArtifactDockerCli(IPackageContext context,
    Func<string[], int, CancellationToken, string?, Task<(int ExitCode, string Output, bool TimedOut, bool WasTruncated)>> run) : DockerCliRunner(context)
{
    internal override Task<DockerCliResolution> ResolveExecutableAsync(CancellationToken cancellationToken = default)
        => Task.FromResult(new DockerCliResolution(Path.Combine(AppContext.BaseDirectory, "controlled-docker"), [], null));
    protected override Task<string> ResolveEndpointAsync(CancellationToken cancellationToken)
        => Task.FromResult("unix:///var/run/sunder-artifact-docker.sock");
    protected override async Task<DockerCliRunResult> RunCoreAsync(IReadOnlyList<string> args, int timeoutSeconds,
        CancellationToken cancellationToken, string? standardInput = null, IProgress<string>? progress = null)
    {
        var logical = args.Count >= 2 && args[0] == "--host" ? args.Skip(2).ToArray() : args.ToArray();
        var result = await run(logical, timeoutSeconds, cancellationToken, standardInput);
        return new(result.ExitCode, result.Output, result.TimedOut, result.WasTruncated);
    }
}

public sealed class ArtifactImageCatalog
{
    public int Count { get; set; } = 2500;
    public Task<object> CreateAsync(IPackageContext context, DockerCliRunner runner, CancellationToken cancellationToken = default)
    {
        cancellationToken.ThrowIfCancellationRequested();
        // Isolate transport content limits from the smaller Host state-write limit.
        // Individual values stay within catalog normalization bounds; no oversized
        // write is admitted or claimed to work in production storage.
        var repository = string.Join('/', Enumerable.Repeat(new string('a', 140), 3));
        var document = JsonSerializer.Serialize(new { schemaVersion = 3, revision = 1,
            images = Enumerable.Range(0, Count).Select(index => new DockerImageDefinition(
                $"example.com/{repository}/image:{index}", DockerImageStatus.Ready, null, new string('x', 512))) },
            new JsonSerializerOptions(JsonSerializerDefaults.Web));
        var state = DispatchProxy.Create<IPackageKeyValueStore, CatalogProjectionState>();
        ((CatalogProjectionState)state).Inner = context.Storage.State;
        ((CatalogProjectionState)state).Document = document;
        var host = DispatchProxy.Create<IPackageContext, CatalogProjectionContext>();
        ((CatalogProjectionContext)host).Inner = context;
        ((CatalogProjectionContext)host).Storage = new CatalogProjectionStorage(context.Storage, state);
        return Task.FromResult<object>(new DockerImageCatalogService(host, runner));
    }
}

public class CatalogProjectionState : DispatchProxy
{
    public IPackageKeyValueStore Inner { get; set; } = null!;
    public string Document { get; set; } = "";
    protected override object? Invoke(MethodInfo? method, object?[]? arguments)
    {
        if (Equals(arguments?[0], "docker.images.v1"))
        {
            if (method!.Name != "GetValueAsync") throw new InvalidOperationException("The projection fixture must not write its synthetic catalog.");
            ((CancellationToken)arguments![1]!).ThrowIfCancellationRequested();
            return Task.FromResult<string?>(Document);
        }
        return method!.Invoke(Inner, arguments);
    }
}

public sealed class CatalogProjectionStorage(IPackageStorageContext inner, IPackageKeyValueStore state) : IPackageStorageContext
{
    public IPackageKeyValueStore State => state;
    public IPackageFileStore Files => inner.Files;
    public IPackageRoleLocalWorkspace RoleLocalWorkspace => inner.RoleLocalWorkspace;
}

public class CatalogProjectionContext : DispatchProxy
{
    public IPackageContext Inner { get; set; } = null!;
    public IPackageStorageContext Storage { get; set; } = null!;
    protected override object? Invoke(MethodInfo? method, object?[]? arguments)
        => method!.Name == "get_Storage" ? Storage : method.Invoke(Inner, arguments);
}
