using System.Text.Json;
using System.Text.Json.Serialization;
using Sunder.Generated.Packages.Sunder_Package_Agent_Models;
using Sunder.Generated.Packages.Sunder_Package_Agent_Services;
using Sunder.Sdk.Rpc;

public sealed class RoleLocalWorkspaceMigrator(
    string contributorId,
    Func<string, bool> canMigrate,
    Func<string, CancellationToken, Task<string>> paths,
    Func<string, CancellationToken, Task> complete) : IWorkspacePathMigratorService
{
    private static readonly JsonSerializerOptions Json = new(JsonSerializerDefaults.Web)
    {
        DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull,
        Converters = { new SunderRpcJsonConverterFactory(), new JsonStringEnumConverter(JsonNamingPolicy.KebabCaseLower, allowIntegerValues: false) },
    };
    public ValueTask<AgentServiceIdentity> DescribeAsync(CancellationToken token = default)
    {
        token.ThrowIfCancellationRequested();
        return ValueTask.FromResult(new AgentServiceIdentity { Id = contributorId, DisplayName = contributorId });
    }
    public ValueTask<bool> CanMigrateAsync(AgentWorkspacePathMigrationContext context, CancellationToken token = default)
    {
        token.ThrowIfCancellationRequested();
        return ValueTask.FromResult(canMigrate(JsonSerializer.Serialize(context, Json)));
    }
    public async ValueTask<IReadOnlyList<AgentWorkspacePathMigrationItem>> ListLegacyPathsAsync(
        AgentWorkspacePathMigrationContext context, CancellationToken token = default)
        => JsonSerializer.Deserialize<AgentWorkspacePathMigrationItem[]>(await paths(JsonSerializer.Serialize(context, Json), token), Json)!;
    public ValueTask CompleteAsync(AgentWorkspacePathMigrationContext context, CancellationToken token = default)
        => new(complete(JsonSerializer.Serialize(context, Json), token));
}
