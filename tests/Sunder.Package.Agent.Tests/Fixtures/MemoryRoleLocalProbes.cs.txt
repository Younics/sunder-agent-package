using System.Text.Json;
using System.Text.Json.Serialization;
using Sunder.Package.Agent.Contracts.Contracts;
using Sunder.Package.Agent.Contracts.Models;
using Sunder.Sdk.Rpc;

public sealed class RoleLocalLifecycleProbe(string observerId, string displayName, Func<string, CancellationToken, ValueTask> handle)
    : IAgentDurableLifecycleObserver
{
    private static readonly JsonSerializerOptions Json = new(JsonSerializerDefaults.Web)
    {
        DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull,
        Converters = { new SunderRpcJsonConverterFactory(), new JsonStringEnumConverter(JsonNamingPolicy.KebabCaseLower, allowIntegerValues: false) },
    };
    public string ObserverId => observerId;
    public string DisplayName => displayName;
    public ValueTask HandleDurableLifecycleEventAsync(AgentDurableLifecycleEventEnvelope value, CancellationToken token = default)
        => handle(JsonSerializer.Serialize(value, Json), token);
}

public sealed class RoleLocalProfileCapabilityProbe(string consumerId, string displayName, string capabilities)
    : IAgentProfileCapabilityConsumer
{
    private readonly AgentProfileCapabilityConsumerDescriptor[] _capabilities = JsonSerializer.Deserialize<AgentProfileCapabilityConsumerDescriptor[]>(
        capabilities, new JsonSerializerOptions(JsonSerializerDefaults.Web))!;
    public string ConsumerId => consumerId;
    public string DisplayName => displayName;
    public IReadOnlyList<AgentProfileCapabilityConsumerDescriptor> ListConsumedCapabilities() => _capabilities;
}
