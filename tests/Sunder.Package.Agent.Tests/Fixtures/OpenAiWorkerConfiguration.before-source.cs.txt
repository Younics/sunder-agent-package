using Microsoft.Extensions.DependencyInjection;
using Sunder.Package.Agent.Protocol;
using Sunder.Package.Agent.Provider.OpenAI.Auth;
using Sunder.Package.Agent.Provider.OpenAI.Transport;
using Sunder.Package.Agent.Provider.Shared;
using Sunder.Sdk.Abstractions;
using Sunder.Sdk.Rpc;
using Sunder.Sdk.Worker;

namespace Sunder.Package.Agent.Provider.OpenAI.Worker;

internal static class OpenAiWorkerConfiguration
{
    internal static SunderWorkerOptions Create(SunderWorkerContext context)
        => CreateCore(context);

    internal static SunderWorkerOptions CreateCore(IPackageContext context)
    {
        var services = new ServiceCollection();
        services.AddSingleton(context);
        services.AddSingleton(new ProviderCredentialAccessor(
            context.Secrets,
            OpenAiProviderConfiguration.ApiKeySecretKey));
        services.AddSingleton<ApiKeyAuthStrategy>();
        services.AddSingleton<CodexConnectedAuthStrategy>();
        services.AddSingleton(_ => CodexHttpClientFactory.CreateBackendClient());
        services.AddSingleton<CodexResponseContinuationStore>();
        services.AddSingleton<CodexConnectedTransport>();
        services.AddSingleton<OpenAiPackageAuthHandler>();
        services.AddSingleton<OpenAiAuthOperationHandler>();
        services.AddSingleton<OpenAiAgentProvider>();
        services.AddSingleton<OpenAiEmbeddingProvider>();
        services.AddSingleton<OpenAiManagementRpcProvider>();

        var serviceProvider = services.BuildServiceProvider(new ServiceProviderOptions
        {
            ValidateOnBuild = true,
            ValidateScopes = true,
        });
        var descriptor = OpenAiManagementProtocol.Descriptor;

        return new SunderWorkerOptions(
        [
            Register(
                "openai.chat",
                AgentRpcContractIds.ChatProvider,
                AgentChatProviderRpc.CreateHandler(
                    serviceProvider.GetRequiredService<OpenAiAgentProvider>())),
            Register(
                "openai.embedding",
                AgentRpcContractIds.EmbeddingProvider,
                AgentEmbeddingProviderRpc.CreateHandler(
                    serviceProvider.GetRequiredService<OpenAiEmbeddingProvider>())),
            Register(
                "openai.continuation.cleaner",
                AgentRpcContractIds.SessionCleaner,
                AgentSessionCleanerRpc.CreateHandler(
                    serviceProvider.GetRequiredService<CodexResponseContinuationStore>())),
            new SunderWorkerProviderRegistration(
                OpenAiManagementProtocol.ProviderId,
                descriptor.ContractId,
                descriptor.Version,
                descriptor.Sha256,
                descriptor.AbiSha256 ?? throw new InvalidDataException(
                    "The OpenAI management RPC descriptor is missing its ABI identity."),
                new Protocol.Generated.SunderPackageAgentProviderOpenaiManagementRpcHandler(
                    serviceProvider.GetRequiredService<OpenAiManagementRpcProvider>())),
        ])
        {
            Composition = new SunderWorkerComposition(serviceProvider),
            SettingsSchema = OpenAiProviderConfiguration.Schema,
            AuthHandler = serviceProvider.GetRequiredService<OpenAiPackageAuthHandler>(),
        };
    }

    private static SunderWorkerProviderRegistration Register(
        string providerId,
        string contractId,
        ISunderRpcServiceHandler handler)
    {
        var descriptor = AgentRpcContractDescriptors.Get(contractId);
        return new SunderWorkerProviderRegistration(
            providerId,
            descriptor.ContractId,
            descriptor.Version,
            descriptor.Sha256,
            descriptor.AbiSha256 ?? throw new InvalidOperationException(
                $"Agent RPC contract '{descriptor.ContractId}' does not have an ABI identity."),
            handler);
    }
}
