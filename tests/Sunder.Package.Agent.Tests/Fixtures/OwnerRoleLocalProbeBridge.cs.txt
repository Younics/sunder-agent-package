// Compiled separately against the fresh Worker. Only BCL values and SDK invocation
// context cross to the retained behavior probe; no legacy model assembly is referenced.
internal sealed class OwnerRoleLocalProbeBridge(
    Func<string, string, SunderRpcInvocationContext?, CancellationToken, ValueTask<(string? Value, string? Error)>> invoke,
    Func<string, string, CancellationToken, IAsyncEnumerable<(string? Value, string? Error)>> subscribe) : IAgentOwnerRuntimeProvider
{
    private static readonly JsonSerializerOptions Json = new(JsonSerializerDefaults.Web)
    { DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull, Converters = { new SunderRpcJsonConverterFactory() } };

    private async ValueTask<T> Call<T>(object request, CancellationToken token, SunderRpcInvocationContext? context = null, [CallerMemberName] string method = "")
    {
        var result = await invoke(method, JsonSerializer.Serialize(request, request.GetType(), Json), context, token);
        if (result.Error is not null) throw new AgentOwnerOperationException(result.Error);
        return JsonSerializer.Deserialize<T>(result.Value!, Json)!;
    }

    private async IAsyncEnumerable<T> Stream<T>(object request, [EnumeratorCancellation] CancellationToken token, [CallerMemberName] string method = "")
    {
        await foreach (var result in subscribe(method, JsonSerializer.Serialize(request, request.GetType(), Json), token).WithCancellation(token))
        {
            if (result.Error is not null) throw new AgentOwnerOperationException(result.Error);
            yield return JsonSerializer.Deserialize<T>(result.Value!, Json)!;
        }
    }
