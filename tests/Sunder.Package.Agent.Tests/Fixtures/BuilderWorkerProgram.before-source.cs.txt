using Sunder.Package.Agent.Builder;
using Sunder.Package.Agent.Builder.Protocol;
using Sunder.Package.Agent.Builder.Protocol.Generated;
using Sunder.Package.Agent.Protocol;
using Sunder.Sdk.Rpc;
using Sunder.Sdk.Worker;

AgentRpcCatalog? ownedCatalog = null;
try
{
    await SunderWorker.RunAsync(context =>
    {
        var catalog = AgentRpcCatalog.CreateDormant(context.Rpc);
        ownedCatalog = catalog;
        var provider = new BuilderRpcProvider(catalog);
        var descriptor = BuilderRpcProtocol.Descriptor;
        return new SunderWorkerOptions(
        [
            new SunderWorkerProviderRegistration(
                "builder.runtime",
                descriptor.ContractId,
                descriptor.Version,
                descriptor.Sha256,
                descriptor.AbiSha256 ?? throw new InvalidOperationException(
                    $"RPC contract '{descriptor.ContractId}' does not have an ABI identity."),
                new SunderPackageAgentBuilderRuntimeRpcHandler(provider)),
        ])
        {
            OnActivated = (_, _) =>
            {
                catalog.StartWatching();
                return ValueTask.CompletedTask;
            },
            OnShutdown = (_, _) =>
            {
                catalog.Dispose();
                if (ReferenceEquals(ownedCatalog, catalog)) ownedCatalog = null;
                return ValueTask.CompletedTask;
            },
        };
    });
}
finally
{
    ownedCatalog?.Dispose();
}
